## 🔑 Key Functions & AI Models

The intelligence of Rananisar AI is driven by a few core JavaScript functions and robust integration with leading AI models.

*   **GPT-4** (OpenAI)

### Core Logic Functions:

The existing `README.md` highlighted the following key JavaScript functions, which underpin the application's interactive and intelligent behavior:

*   `generateNewQuiz()`: Responsible for initiating the AI-powered generation of new quiz questions and content.

    const toast =
      document.getElementById("toast");

    const moreBtn =

    const moreMenu =
      document.getElementById("moreMenu");

    const floatingTools =
      document.getElementById("floatingTools");


    /* ---------------------------------------------------------
       STATE
    --------------------------------------------------------- */

    let selectedTemplate = "Cinematic Portrait";
    let videoMode = false;
    let currentRatio = "2:3";

    /* ---------------------------------------------------------
       TOAST
    --------------------------------------------------------- */

    function showToast(message) {

      toast.textContent = message;

      toast.classList.remove("opacity-0");

      clearTimeout(toastTimer);

      toastTimer = setTimeout(() => {

        toast.classList.add("opacity-0");

      }, 1800);
    }

       PHOTO UPLOAD
    --------------------------------------------------------- */

    uploadBtn.addEventListener("click", () => {

      photoInput.click();

    });


    photoInput.addEventListener("change", event => {

      const files = [...event.target.files];

      if (!files.length) return;

      files.forEach(file => {

        if (!file.type.startsWith("image/")) {
          return;
        }

        const reader = new FileReader();

        reader.onload = e => {

          const wrapper =
            document.createElement("div");

          wrapper.className =
            "photo-item relative shrink-0 w-[82px] h-[82px] rounded-2xl overflow-hidden border border-white/10 fade-in";

          wrapper.innerHTML = `

            <img
              src="${e.target.result}"
              class="photo-preview"
              alt="Uploaded photo"
            >

            <button
              class="remove-photo absolute top-1 right-1 w-6 h-6 rounded-full bg-black/70 text-white flex items-center justify-center"
            >
              ×
            </button>

          `;

          const removeButton =
            wrapper.querySelector(".remove-photo");

          removeButton.addEventListener(
            "click",
            () => {

              wrapper.remove();

              showToast("Photo removed");

            }
          );

          photoStrip.appendChild(wrapper);

        reader.readAsDataURL(file);

                "Copied to clipboard"
              );

            } catch {

              showToast(
                "Sharing unavailable"
              );

            }

          }

        }
      );

       INITIAL STATE

    updateCanvasRatio();
